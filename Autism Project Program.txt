def preprocess_images():
    images = []
    processed_images = []

    for filename in os.listdir(DATASET_FOLDER):
        if filename.lower().endswith(('.png', '.jpg', '.jpeg')):
            # Load image
            image_path = os.path.join(DATASET_FOLDER, filename)
            img = Image.open(image_path)
            
            # Convert to grayscale
            img = img.convert("L")

            # Resize to 224x224
            img = img.resize((224, 224))

            # Convert to numpy array and normalize
            img_array = np.array(img) / 255.0  # Normalizing between 0-1

            # Save processed image
            processed_path = os.path.join(PROCESSED_FOLDER, filename)
            Image.fromarray((img_array * 255).astype(np.uint8)).save(processed_path)

            # Store paths for HTML rendering
            images.append(image_path)
            processed_images.append(processed_path)

    return images, processed_images

def binarize_images():
    images = []
    binary_images = []
    threshold_value = 128  # Adjust as needed

    for filename in os.listdir(DATASET_FOLDER):
        if filename.lower().endswith(('.png', '.jpg', '.jpeg')):
            # Load image
            image_path = os.path.join(DATASET_FOLDER, filename)

            # Convert to numpy array
            img_array = np.array(img)

            images.append(image_path)

SEGMENTED_FOLDER = 'static/segmented/'

# Ensure segmented folder exists
os.makedirs(SEGMENTED_FOLDER, exist_ok=True)

def segment_images():
    images = []
    segmented_images = []

    for filename in os.listdir(DATASET_FOLDER):
        if filename.lower().endswith(('.png', '.jpg', '.jpeg')):
            image_path = os.path.join(DATASET_FOLDER, filename)
            img = cv2.imread(image_path)

            # Convert to grayscale
            gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

            # Apply Otsu's thresholding
            _, thresh = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

            # Use morphological operations to remove small noise
            kernel = np.ones((3,3), np.uint8)
            opening = cv2.morphologyEx(thresh, cv2.MORPH_OPEN, kernel, iterations=2)

            # Background area
            sure_bg = cv2.dilate(opening, kernel, iterations=3)

            # Distance transform and foreground area detection
            dist_transform = cv2.distanceTransform(opening, cv2.DIST_L2, 5)
            _, sure_fg = cv2.threshold(dist_transform, 0.5 * dist_transform.max(), 255, 0)

            # Find unknown region
            sure_fg = np.uint8(sure_fg)
            unknown = cv2.subtract(sure_bg, sure_fg)

            # Label markers
            _, markers = cv2.connectedComponents(sure_fg)
            markers = markers + 1
            markers[unknown == 255] = 0

            # Apply watershed algorithm
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)  # Convert to RGB for PIL
            markers = cv2.watershed(img, markers)
            img[markers == -1] = [255, 0, 0]  # Mark boundaries in red

            # Save segmented image
            segmented_path = os.path.join(SEGMENTED_FOLDER, filename)
            Image.fromarray(img).save(segmented_path)

            # Store paths for HTML display
            images.append(image_path)
            segmented_images.append(segmented_path)

    return images, segmented_images

FEATURES_FOLDER = "static/features"  # Folder for extracted features
os.makedirs(FEATURES_FOLDER, exist_ok=True)  # Ensure folder exists

def extract_features():
    images = []  
    feature_images = []  

    for filename in os.listdir(DATASET_FOLDER):
        img_path = os.path.join(DATASET_FOLDER, filename)
        
        # Load image
        img = cv2.imread(img_path)
        if img is None:
            print(f"Skipping {filename} (Image not found or corrupted)")
            continue  

        print(f"Processing {filename}...")  # Debugging output

        # Convert to grayscale
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

        # 1️⃣ Histogram Equalization
        hist_img = cv2.equalizeHist(gray)

        # 2️⃣ Texture Features (Laplacian)
        texture_img = cv2.Laplacian(gray, cv2.CV_64F)
        texture_img = cv2.convertScaleAbs(texture_img)

        # 3️⃣ Edge Detection (Sobel)
        edges = cv2.Sobel(gray, cv2.CV_64F, 1, 1, ksize=5)
        edges = cv2.convertScaleAbs(edges)

        # Resize for consistency
        hist_resized = cv2.resize(hist_img, (100, 100))
        texture_resized = cv2.resize(texture_img, (100, 100))
        edges_resized = cv2.resize(edges, (100, 100))

        # Stack images vertically
        combined = np.vstack((hist_resized, texture_resized, edges_resized))

        # Save feature image
        feature_path = os.path.join(FEATURES_FOLDER, filename)
        cv2.imwrite(feature_path, combined)

        images.append(img_path)
        feature_images.append(feature_path)

    return images, feature_images, []

# Function to extract features
def extract_features():
    images = []
    labels = []
    feature_vectors = []

    for filename in os.listdir(DATASET_FOLDER):
        img_path = os.path.join(DATASET_FOLDER, filename)

        # Read and preprocess image
        img = cv2.imread(img_path)
        if img is None:
            continue

        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

        # 1️⃣ Histogram Equalization
        hist_img = cv2.equalizeHist(gray)

        # 2️⃣ Texture Features (Laplacian)
        texture_img = cv2.Laplacian(gray, cv2.CV_64F)
        texture_img = cv2.convertScaleAbs(texture_img)

        # 3️⃣ Edge Detection (Sobel)
        edges = cv2.Sobel(gray, cv2.CV_64F, 1, 1, ksize=5)
        edges = cv2.convertScaleAbs(edges)

        # Resize for consistency
        img_resized = cv2.resize(gray, (100, 100)).flatten()
        hist_resized = cv2.resize(hist_img, (100, 100)).flatten()
        texture_resized = cv2.resize(texture_img, (100, 100)).flatten()
        edges_resized = cv2.resize(edges, (100, 100)).flatten()

        # Combine features into a single vector
        feature_vector = np.hstack((img_resized, hist_resized, texture_resized, edges_resized))
        feature_vectors.append(feature_vector)

        # Label extraction (assuming filename contains label, e.g., "cat_01.jpg")
        label = filename.split("_")[0]  # Extract class from filename
        labels.append(label)
        images.append(filename)

    return np.array(feature_vectors), np.array(labels), images

def kmeans_color_quantization(image, clusters=8, rounds=1):
    h, w = image.shape[:2]
    samples = np.zeros([h*w,3], dtype=np.float32)
    count = 0

    for x in range(h):
        for y in range(w):
            samples[count] = image[x][y]
            count += 1

    compactness, labels, centers = cv2.kmeans(samples,
            clusters, 
            None,
            (cv2.TERM_CRITERIA_EPS + cv2.TERM_CRITERIA_MAX_ITER, 10000, 0.0001), 
            rounds, 
            cv2.KMEANS_RANDOM_CENTERS)

    centers = np.uint8(centers)
    res = centers[labels.flatten()]
    return res.reshape((image.shape))

def view_photo():
    ff1=open("photo.txt","w")
    ff1.write("1")

        value = mycursor.fetchall()

    if request.method=='POST':
        print("Training")
        vid=request.form['vid']
        cursor = mydb.cursor()
        cursor.execute("SELECT * FROM vt_face where vid=%s",(vid, ))
        dt = cursor.fetchall()
        for rs in dt:
            ##Preprocess
            path="static/frame/"+rs[2]
            path2="static/process1/"+rs[2]
            mm2 = PIL.Image.open(path).convert('L')
            rz = mm2.resize((200,200), PIL.Image.ANTIALIAS)
            rz.save(path2)
            #noice
            img = cv2.imread('static/process1/'+rs[2]) 
            dst = cv2.fastNlMeansDenoisingColored(img, None, 10, 10, 7, 15)
            fname2='ns_'+rs[2]
            cv2.imwrite("static/process1/"+fname2, dst)
            image = cv2.imread('static/process1/'+rs[2])
            original = image.copy()
            kmeans = kmeans_color_quantization(image, clusters=4)

            # Convert to grayscale, Gaussian blur, adaptive threshold
            gray = cv2.cvtColor(kmeans, cv2.COLOR_BGR2GRAY)
            blur = cv2.GaussianBlur(gray, (3,3), 0)
            thresh = cv2.adaptiveThreshold(blur,255,cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY_INV,21,2)

            # Draw largest enclosing circle onto a mask
            mask = np.zeros(original.shape[:2], dtype=np.uint8)
            cnts = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
            cnts = cnts[0] if len(cnts) == 2 else cnts[1]
            cnts = sorted(cnts, key=cv2.contourArea, reverse=True)
            for c in cnts:
                ((x, y), r) = cv2.minEnclosingCircle(c)
                cv2.circle(image, (int(x), int(y)), int(r), (36, 255, 12), 2)
                cv2.circle(mask, (int(x), int(y)), int(r), 255, -1)
                break
            
            # Bitwise-and for result
            result = cv2.bitwise_and(original, original, mask=mask)
            result[mask==0] = (0,0,0)


            cv2.imwrite("static/process1/bin_"+rs[2], thresh)
            

            ###RPN - Segment
            img = cv2.imread('static/process1/'+rs[2])
            gray = cv2.cvtColor(img,cv2.COLOR_BGR2GRAY)
            ret, thresh = cv2.threshold(gray,0,255,cv2.THRESH_BINARY_INV+cv2.THRESH_OTSU)

            
            kernel = np.ones((3,3),np.uint8)
            opening = cv2.morphologyEx(thresh,cv2.MORPH_OPEN,kernel, iterations = 2)

            # sure background area
            sure_bg = cv2.dilate(opening,kernel,iterations=3)

            # Finding sure foreground area
            dist_transform = cv2.distanceTransform(opening,cv2.DIST_L2,5)
            ret, sure_fg = cv2.threshold(dist_transform,0.7*dist_transform.max(),255,0)

            # Finding unknown region
            sure_fg = np.uint8(sure_fg)
            segment = cv2.subtract(sure_bg,sure_fg)
            img = Image.fromarray(img)
            segment = Image.fromarray(segment)
            path3="static/process2/fg_"+rs[2]
            segment.save(path3)
            ####
            img = cv2.imread('static/process2/fg_'+rs[2])
            gray = cv2.cvtColor(img,cv2.COLOR_BGR2GRAY)
            ret, thresh = cv2.threshold(gray,0,255,cv2.THRESH_BINARY_INV+cv2.THRESH_OTSU)

            
            kernel = np.ones((3,3),np.uint8)
            opening = cv2.morphologyEx(thresh,cv2.MORPH_OPEN,kernel, iterations = 2)

            sure_bg = cv2.dilate(opening,kernel,iterations=3)

            dist_transform = cv2.distanceTransform(opening,cv2.DIST_L2,5)
            ret, sure_fg = cv2.threshold(dist_transform,0.7*dist_transform.max(),255,0)

            sure_fg = np.uint8(sure_fg)
            segment = cv2.subtract(sure_bg,sure_fg)
            img = Image.fromarray(img)
            segment = Image.fromarray(segment)
            path3="static/process2/fg_"+rs[2]
            segment.save(path3)

            image = cv2.imread(path2)
            gray = cv2.cvtColor(image, cv2.COLOR_BGR2GRAY)
            edged = cv2.Canny(gray, 50, 100)
            image = Image.fromarray(image)
            edged = Image.fromarray(edged)
            path4="static/process3/"+rs[2]
            edged.save(path4)
       
        return redirect(url_for('view_photo1',vid=vid))
        
    return render_template('view_photo.html', result=value,vid=vid)

        return render_template('result.html', msg=msg, score=session['score'])

    index = questions_bank.index(q)
